NASM 64-BIT STRING OPERATIONS - ALL 20 THREE-FUNCTION COMBINATIONS (SUBSTRING REMOVED)
Only these six operations: Length, Copy, Reverse, Compare, Concatenate, Palindrome.
Each section is a separate complete program.
Input: String1 and String2 are read before the menu.
Menu: the selected three operations plus Exit.
Assemble each program separately with NASM for Linux x86-64.

    mov byte [rbx + rcx], 0
    mov rax, rcx
    jmp read_return

read_finish:
    mov byte [rbx + rax], 0
    jmp read_return

read_empty:
    xor rax, rax

read_return:
    pop rbx
    ret

; Display unsigned decimal integer
; Input: RAX = number

display:
    lea rsi, [buff + 23]
    mov byte [rsi], 10
    mov rcx, 1
    mov rbx, 10

    test rax, rax
    jnz display_loop

    dec rsi
    mov byte [rsi], '0'
    inc rcx
    jmp display_write

display_loop:
    xor rdx, rdx
    div rbx
    add dl, '0'
    dec rsi
    mov [rsi], dl
    inc rcx
    test rax, rax
    jnz display_loop

display_write:
    mov rdx, rcx
    mov rax, 1
    mov rdi, 1
    syscall
    ret

exit:
    mov rax, 60
    xor rdi, rdi
    syscall
