1. Addition + Subtraction

%macro WRITE 02
    mov rax,01
    mov rdi,01
    mov rsi,%1
    mov rdx,%2
    syscall
%endmacro

%macro READ 02
    mov rax,00
    mov rdi,00
    mov rsi,%1
    mov rdx,%2
    syscall
%endmacro

%macro EXIT 00
    mov rax,60
    mov rdi,00
    syscall
%endmacro

section .data

menu db 10,"1. Addition",10
     db "2. Subtraction",10
     db "3. Exit",10
     db "Enter your choice: "
menulen equ $-menu

msg1 db "Enter 2 numbers",10
len1 equ $-msg1

msg_add db "Addition is ",10
len_add equ $-msg_add

msg_sub db "Subtraction is ",10
len_sub equ $-msg_sub

section .bss

char_buff resb 17
actl      resq 1
choice    resb 2
a         resq 1
b         resq 1
c         resq 1

section .text
global _start

_start:

    WRITE msg1,len1

    call accept
    mov qword[a],rbx

    call accept
    mov qword[b],rbx

printmenu:

    WRITE menu,menulen
    READ choice,2

    cmp byte[choice],31H
    je addition

    cmp byte[choice],32H
    je subtraction

    cmp byte[choice],33H
    je exit

    jmp printmenu

addition:

    mov rbx,qword[a]
    add rbx,qword[b]
    mov qword[c],rbx

    WRITE msg_add,len_add

    mov rbx,qword[c]
    call display

    jmp printmenu

subtraction:

    mov rbx,qword[a]
    sub rbx,qword[b]
    mov qword[c],rbx

    WRITE msg_sub,len_sub

    mov rbx,qword[c]
    call display

    jmp printmenu

exit:
    EXIT

accept:

    READ char_buff,17
    dec rax
    mov qword[actl],rax

    mov rsi,char_buff
    mov rbx,00H

up:

    shl rbx,04H
    mov rdx,00H
    mov dl,byte[rsi]

    cmp dl,39H
    jbe sub30

    sub dl,07H

sub30:

    sub dl,30H
    add rbx,rdx

    inc rsi
    dec qword[actl]

    jnz up

    RET

display:

    mov rsi,char_buff
    mov rcx,16

above:

    rol rbx,04H
    mov dl,bl
    AND dl,0FH

    cmp dl,09H
    jbe add30

    add dl,07H

add30:

    add dl,30H
    mov byte[rsi],dl

    inc rsi
    dec rcx

    jnz above

    WRITE char_buff,16

    RET

    dec rsi

    inc rcx

    cmp rax,00H
    jnz convert

    mov rdx,rcx

    WRITE rsi,rdx

    ret
